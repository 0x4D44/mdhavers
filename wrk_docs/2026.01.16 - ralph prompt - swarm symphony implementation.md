// Vector operations for 3D vectors [x, y, z]
dae vec_add(a, b) {
    gie [a[0] + b[0], a[1] + b[1], a[2] + b[2]]
}

dae vec_sub(a, b) {
    gie [a[0] - b[0], a[1] - b[1], a[2] - b[2]]
}

dae vec_scale(v, s) {
    gie [v[0] * s, v[1] * s, v[2] * s]
}

dae vec_magnitude(v) {
    gie sqrt(v[0] * v[0] + v[1] * v[1] + v[2] * v[2])
}

dae vec_normalize(v) {
    ken m = vec_magnitude(v)
    gin m < 0.0001 { gie [0, 0, 0] }
    gie vec_scale(v, 1.0 / m)
}

dae vec_distance(a, b) {
    gie vec_magnitude(vec_sub(a, b))
}

dae clamp_magnitude(v, max_mag) {
    ken m = vec_magnitude(v)
    gin m > max_mag {
        gie vec_scale(vec_normalize(v), max_mag)
    }
    gie v
}

dae vec_dot(a, b) {
    gie a[0] * b[0] + a[1] * b[1] + a[2] * b[2]
}
```

ken CELL_SIZE = 5.0

dae spatial_key(x, y, z) {
    ken cx = floor(x / CELL_SIZE)
    ken cy = floor(y / CELL_SIZE)
    ken cz = floor(z / CELL_SIZE)
    gie f"{cx},{cy},{cz}"
}

dae spatial_insert(spatial, agent) {
    ken key = spatial_key(agent["pos"][0], agent["pos"][1], agent["pos"][2])
    gin spatial[key] == naething {
        spatial[key] = []
    }
    push(spatial[key], agent["id"])
}

dae spatial_query(spatial, agents_map, pos, radius) {
    ken results = []
    ken cells = ceil(radius / CELL_SIZE)
    ken base_x = floor(pos[0] / CELL_SIZE)
    ken base_y = floor(pos[1] / CELL_SIZE)
    ken base_z = floor(pos[2] / CELL_SIZE)

    fer dx in range(-cells, cells + 1) {
        fer dy in range(-cells, cells + 1) {
            fer dz in range(-cells, cells + 1) {
                ken key = f"{base_x + dx},{base_y + dy},{base_z + dz}"
                gin spatial[key] != naething {
                    ilk(spatial[key], |id| {
                        ken agent = agents_map[id]
                        gin agent != naething {
                            ken dist = vec_distance(pos, agent["pos"])
                            gin dist <= radius {
                                push(results, agent)
                            }
                        }
                    })
                }
            }
        }
    }
    gie results
}
```

### 1.4 Boid Behaviors

**Add to `simulation.braw`:**
```braw
dae calculate_separation(agent, neighbors, rules) {
    ken force = [0.0, 0.0, 0.0]
    ken count = 0
    ilk(neighbors, |n| {
        gin n["id"] != agent["id"] {
            ken dist = vec_distance(agent["pos"], n["pos"])
            gin dist < rules["separation_radius"] an dist > 0.001 {
                ken diff = vec_sub(agent["pos"], n["pos"])
                diff = vec_scale(vec_normalize(diff), 1.0 / dist)
                force = vec_add(force, diff)
                count = count + 1
            }
        }
    })
    gin count > 0 {
        gie vec_normalize(force)
    }
    gie force
}

dae calculate_alignment(agent, neighbors, rules) {
    ken avg_vel = [0.0, 0.0, 0.0]
    ken count = 0
    ilk(neighbors, |n| {
        gin n["id"] != agent["id"] {
            ken dist = vec_distance(agent["pos"], n["pos"])
            gin dist < rules["alignment_radius"] {
                avg_vel = vec_add(avg_vel, n["vel"])
                count = count + 1
            }
        }
    })
    gin count > 0 {
        avg_vel = vec_scale(avg_vel, 1.0 / count)
        gie vec_normalize(vec_sub(avg_vel, agent["vel"]))
    }
    gie [0.0, 0.0, 0.0]
}

dae calculate_cohesion(agent, neighbors, rules) {
    ken center = [0.0, 0.0, 0.0]
    ken count = 0
    ilk(neighbors, |n| {
        gin n["id"] != agent["id"] {
            ken dist = vec_distance(agent["pos"], n["pos"])
            gin dist < rules["cohesion_radius"] {
                center = vec_add(center, n["pos"])
                count = count + 1
            }
        }
    })
    gin count > 0 {
        center = vec_scale(center, 1.0 / count)
        gie vec_normalize(vec_sub(center, agent["pos"]))
    }
    gie [0.0, 0.0, 0.0]
}

dae calculate_avoidance(agent, world) {
    ken force = [0.0, 0.0, 0.0]
    ken predators = sieve(world["agents"], |a| a["kind"] == "predator")
    ilk(predators, |p| {
        ken dist = vec_distance(agent["pos"], p["pos"])
        gin dist < world["rules"]["fear_radius"] an dist > 0.001 {
            ken flee = vec_sub(agent["pos"], p["pos"])
            flee = vec_scale(vec_normalize(flee), 1.0 / (dist * dist))
            force = vec_add(force, flee)
        }
    })
    gie vec_normalize(force)
}

dae calculate_boundary(agent, bounds) {
    ken force = [0.0, 0.0, 0.0]
    ken margin = 5.0

    fer i in range(3) {
        gin agent["pos"][i] < bounds[i] + margin {
            force[i] = (margin - (agent["pos"][i] - bounds[i])) / margin
        }
        gin agent["pos"][i] > bounds[i + 3] - margin {
            force[i] = -((agent["pos"][i] - (bounds[i + 3] - margin))) / margin
        }
    }
    gie force
}
```

**Done when:** Boids flock together, flee from predators, stay in bounds.

### 1.5 Simulation Loop

**Add to `simulation.braw`:**
```braw
ken FIXED_DT = 1.0 / 60.0

dae update_agents(world) {
    // Rebuild spatial hash
    world["spatial"] = {}
    ilk(world["agents"], |a| spatial_insert(world["spatial"], a))

    ken events = []
    ken max_radius = max(world["rules"]["separation_radius"],
                         max(world["rules"]["alignment_radius"],
                             world["rules"]["cohesion_radius"]))

    ilk(world["agents"], |agent| {
        keek agent["kind"] {
            whan "boid" -> {
                ken neighbors = spatial_query(world["spatial"], world["agents_map"],
                                              agent["pos"], max_radius)

                ken sep = calculate_separation(agent, neighbors, world["rules"])
                ken ali = calculate_alignment(agent, neighbors, world["rules"])
                ken coh = calculate_cohesion(agent, neighbors, world["rules"])
                ken avoid = calculate_avoidance(agent, world)
                ken bound = calculate_boundary(agent, world["bounds"])

                ken force = vec_add(
                    vec_scale(sep, world["rules"]["separation_weight"]),
                    vec_add(vec_scale(ali, world["rules"]["alignment_weight"]),
                        vec_add(vec_scale(coh, world["rules"]["cohesion_weight"]),
                            vec_add(vec_scale(avoid, world["rules"]["fear_weight"]),
                                bound))))

                force = clamp_magnitude(force, world["rules"]["max_force"])
                agent["vel"] = clamp_magnitude(
                    vec_add(agent["vel"], vec_scale(force, FIXED_DT)),
                    world["rules"]["max_speed"])
                agent["pos"] = vec_add(agent["pos"], vec_scale(agent["vel"], FIXED_DT))
            }
            whan "predator" -> {
                // Chase nearest boid
                ken boids = sieve(world["agents"], |a| a["kind"] == "boid")
                gin len(boids) > 0 {
                    ken nearest = boids[0]
                    ken nearest_dist = vec_distance(agent["pos"], nearest["pos"])
                    ilk(boids, |b| {
                        ken d = vec_distance(agent["pos"], b["pos"])
                        gin d < nearest_dist {
                            nearest = b
                            nearest_dist = d
                        }
                    })
                    ken chase = vec_normalize(vec_sub(nearest["pos"], agent["pos"]))
                    agent["vel"] = clamp_magnitude(
                        vec_add(agent["vel"], vec_scale(chase, 0.03)),
                        world["rules"]["max_speed"] * 0.8)
                }
                agent["pos"] = vec_add(agent["pos"], vec_scale(agent["vel"], FIXED_DT))

                // Boundary
                ken bound = calculate_boundary(agent, world["bounds"])
                agent["vel"] = vec_add(agent["vel"], vec_scale(bound, 0.1))
            }
        }

        // Decrement sound cooldown
        gin agent["sound_cooldown"] > 0 {
            agent["sound_cooldown"] = agent["sound_cooldown"] - 1
        }
    })

    world["tick"] = world["tick"] + 1
    gie events
}

dae calculate_coherence(world) {
    ken boids = sieve(world["agents"], |a| a["kind"] == "boid")
    gin len(boids) < 2 { gie 1.0 }

    ken avg_pos = [0.0, 0.0, 0.0]
    ilk(boids, |b| { avg_pos = vec_add(avg_pos, b["pos"]) })
    avg_pos = vec_scale(avg_pos, 1.0 / len(boids))

    ken total_dist = 0.0
    ilk(boids, |b| { total_dist = total_dist + vec_distance(b["pos"], avg_pos) })

    ken avg_dist = total_dist / len(boids)
    ken max_dist = world["bounds"][3] - world["bounds"][0]
    gie max(0.0, 1.0 - (avg_dist / (max_dist * 0.5)))
}
```

**Done when:** 500 boids simulate smoothly, predators chase, simulation is deterministic.
