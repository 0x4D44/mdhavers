## Overview

This plan implements full Windows platform support for mdhavers in two workstreams:

1. **Workstream A: Unified Audio Abstraction** (8-11 hours)
2. **Workstream B: LLVM Runtime Windows Port** (14-21 hours)

**Recommended execution order:** Workstream A first (faster, unblocks Windows builds immediately).

// runtime/platform/platform_win32.c
#include <windows.h>

struct mdh_dir {
    HANDLE handle;
    WIN32_FIND_DATAW find_data;
    int first_read;
    mdh_dirent_t entry;
};

mdh_dir_t* mdh_opendir(const char* path) {
    mdh_dir_t* dir = malloc(sizeof(mdh_dir_t));
    wchar_t wpath[MAX_PATH];
    MultiByteToWideChar(CP_UTF8, 0, path, -1, wpath, MAX_PATH);
    wcscat(wpath, L"\\*");  // Append wildcard

    dir->handle = FindFirstFileW(wpath, &dir->find_data);
    if (dir->handle == INVALID_HANDLE_VALUE) {
        free(dir);
        return NULL;
    }
    dir->first_read = 1;
    return dir;
}
```

mdh_process_t* mdh_spawn(const char* program, char* const argv[],
                         int stdin_fd, int stdout_fd, int stderr_fd) {
    STARTUPINFOW si = { sizeof(si) };
    PROCESS_INFORMATION pi;

    si.dwFlags = STARTF_USESTDHANDLES;
    si.hStdInput = (HANDLE)_get_osfhandle(stdin_fd);
    si.hStdOutput = (HANDLE)_get_osfhandle(stdout_fd);
    si.hStdError = (HANDLE)_get_osfhandle(stderr_fd);

    // Build command line from argv
    wchar_t cmdline[32768];
    build_command_line(argv, cmdline, sizeof(cmdline)/sizeof(wchar_t));

    if (!CreateProcessW(NULL, cmdline, NULL, NULL, TRUE, 0, NULL, NULL, &si, &pi)) {
        return NULL;
    }

    mdh_process_t* proc = malloc(sizeof(mdh_process_t));
    proc->handle = pi.hProcess;
    proc->thread = pi.hThread;
    return proc;
}
```

int mdh_terminal_get_size(int* width, int* height) {
    CONSOLE_SCREEN_BUFFER_INFO csbi;
    HANDLE hConsole = GetStdHandle(STD_OUTPUT_HANDLE);

    if (!GetConsoleScreenBufferInfo(hConsole, &csbi)) {
        return -1;
    }

    *width = csbi.srWindow.Right - csbi.srWindow.Left + 1;
    *height = csbi.srWindow.Bottom - csbi.srWindow.Top + 1;
    return 0;
}

int mdh_terminal_raw_mode(mdh_terminal_state_t* old_state) {
    HANDLE hConsole = GetStdHandle(STD_INPUT_HANDLE);
    GetConsoleMode(hConsole, &old_state->mode);

    DWORD new_mode = old_state->mode;
    new_mode &= ~(ENABLE_LINE_INPUT | ENABLE_ECHO_INPUT);
    new_mode |= ENABLE_VIRTUAL_TERMINAL_INPUT;

    return SetConsoleMode(hConsole, new_mode) ? 0 : -1;
}
```

static int wsa_initialized = 0;

int mdh_net_init(void) {
    if (wsa_initialized) return 0;

    WSADATA wsaData;
    int result = WSAStartup(MAKEWORD(2, 2), &wsaData);
    if (result != 0) {

    if (wsa_initialized) {

struct mdh_thread {
    HANDLE handle;
    void* (*func)(void*);
    void* arg;
    void* result;
};

static DWORD WINAPI thread_wrapper(LPVOID arg) {
    mdh_thread_t* thread = (mdh_thread_t*)arg;
    thread->result = thread->func(thread->arg);
    return 0;
}

mdh_thread_t* mdh_thread_create(void* (*func)(void*), void* arg) {
    mdh_thread_t* thread = malloc(sizeof(mdh_thread_t));
    thread->func = func;
    thread->arg = arg;
    thread->result = NULL;

    thread->handle = CreateThread(NULL, 0, thread_wrapper, thread, 0, NULL);
    if (thread->handle == NULL) {
        free(thread);
        return NULL;
    }
    return thread;
}

struct mdh_mutex {
    SRWLOCK lock;
};

mdh_mutex_t* mdh_mutex_create(void) {
    mdh_mutex_t* mutex = malloc(sizeof(mdh_mutex_t));
    InitializeSRWLock(&mutex->lock);
    return mutex;
}

void mdh_mutex_lock(mdh_mutex_t* mutex) {
    AcquireSRWLockExclusive(&mutex->lock);
}
```

uint64_t mdh_time_monotonic_ns(void) {
    static LARGE_INTEGER frequency = {0};
    if (frequency.QuadPart == 0) {
        QueryPerformanceFrequency(&frequency);
    }

    LARGE_INTEGER counter;
    QueryPerformanceCounter(&counter);

    return (uint64_t)((counter.QuadPart * 1000000000ULL) / frequency.QuadPart);
}

void mdh_sleep_ms(uint32_t milliseconds) {
    Sleep(milliseconds);
}
